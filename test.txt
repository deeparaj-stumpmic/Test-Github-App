import requests

def get_user_data(user_id):
    url = f"https://api.example.com/users/{user_id}"

    # No timeout
    response = requests.get(url)

    # No status code validation
    data = response.json()

    # Logging sensitive data
    print(f"Retrieved user data: {data}")

    return data
