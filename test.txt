import os
import traceback
from datetime import datetime, timezone

from google.cloud import firestore

from app.core.config import settings

# -------------------------------
# Firestore Service
# -------------------------------


class FirestoreService:
    def __init__(self):
        self.client = None
        self._init_client()

    def _init_client(self):
        emulator_host = settings.FIRESTORE_EMULATOR_HOST

        if emulator_host:
            # LOCAL MODE: Uses the emulator
            os.environ["FIRESTORE_EMULATOR_HOST"] = emulator_host
            self.client = firestore.Client(
                project=settings.PROJECT_ID or "demo-project"
            )
            print(
                f"DEBUG: Connected to Firestore Emulator at {emulator_host} (project={settings.PROJECT_ID})"
            )
        else:
            # PRODUCTION MODE: Auto-detects credentials via ADC / Cloud Run service account
            try:
                self.client = firestore.Client(project=settings.PROJECT_ID)
                print(
                    f"DEBUG: Firestore initialized for project: {self.client.project}"
                )
            except Exception as e:
                print(f"ERROR: Failed to initialize Firestore: {e}")
                traceback.print_exc()
                self.client = None

    def is_available(self) -> bool:
        return self.client is not None

    def create_demo_request(self, form_data: dict):
        if not self.is_available():
            # Providing a more descriptive error for logs
            raise RuntimeError(
                "Firestore client not available. Verify Service Account roles/datastore.user is assigned."
            )

        # Ensure all data is Firestore-serializable
        clean_data = {
            k: (v if not isinstance(v, (list, tuple)) else list(v))
            for k, v in form_data.items()
            if v is not None
        }

        try:
            doc_ref = self.client.collection("demo_requests").document()
            doc_ref.set(
                {
                    **clean_data,
                    "created_at": datetime.now(timezone.utc),
                    "status": "pending",
                    "source": "website_form",
                    "environment": (
                        "emulator" if settings.FIRESTORE_EMULATOR_HOST else "production"
                    ),
                }
            )
            return doc_ref.id
        except Exception:
            print("ERROR: Failed to write to Firestore collection 'demo_requests'")
            traceback.print_exc()
            raise


# -------------------------------
# Singleton accessor
# -------------------------------
_db_instance = None


def get_db():
    global _db_instance
    if _db_instance is None:
        _db_instance = FirestoreService()
    return _db_instance
